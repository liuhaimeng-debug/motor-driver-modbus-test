u16 crc16(u8 *addr, u8 Len)
{
    u16 crc = 0xffff;//CRC寄存器

    for(u8 num=0; num<Len; num++)
    {
        crc = (*addr++)^crc;                //把数据与16位的CRC寄存器的低8位相异或，结果存放于CRC寄存器。
        for(u8 x=0;x<8;x++)                 //循环8次
        {
            if(crc&0x0001)                  //判断最低位为：“1”
            {
                crc = crc>>1;               //先右移
                crc = crc^0xA001;           //再与0xA001异或
            }
            else                           //判断最低位为：“0”
            {
                crc = crc>>1; //右移
            }
        }
    }
    return crc;//返回CRC校验值
}
